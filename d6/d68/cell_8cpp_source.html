762void CGeomCell::CalcAllLayerElevsAndD50(void)
763{
764 m_VdAllHorizonTopElev.clear();
765 m_VdAllHorizonTopElev.push_back(m_dBasementElevation); // Elevation of top of the basement
766
767 // Calculate the elevation of the top of all other layers
768 int m = 0;
769
770 for (unsigned int n = 0; n < m_VLayerAboveBasement.size(); n++)
771 m_VdAllHorizonTopElev.push_back(m_VLayerAboveBasement[n].dGetTotalThickness() + m_VdAllHorizonTopElev[m++]); // Elevation of top of layer n, ignoring any talus
772
773 // Now calculate the d50 of the topmost unconsolidated sediment layer with non-zero thickness. If there is no unconsolidated sediment, m_dUnconsD50 is set to DBL_NODATA
774 m_dUnconsD50 = DBL_NODATA;
775
776 for (int n = static_cast<int>(m_VLayerAboveBasement.size()) - 1; n >= 0; n--)
777 {
778 double const dUnconsThick = m_VLayerAboveBasement[n].dGetAllUnconsDepth();
779
780 if (dUnconsThick > 0)
781 {
782 // This is a layer with non-zero thickness of unconsolidated sediment
783 CRWCellSediment const* pUnconsSedLayer = m_VLayerAboveBasement[n].pGetUnconsolidatedSediment();
784 double const dFineProp = pUnconsSedLayer->dGetFineDepth() / dUnconsThick;
785 double const dSandProp = pUnconsSedLayer->dGetSandDepth() / dUnconsThick;
786 double const dCoarseProp = pUnconsSedLayer->dGetCoarseDepth() / dUnconsThick;
787
788 // Calculate d50 for the unconsolidated sediment
789 m_dUnconsD50 = (dFineProp * m_pGrid->pGetSim()->dGetD50Fine()) + (dSandProp * m_pGrid->pGetSim()->dGetD50Sand()) + (dCoarseProp * m_pGrid->pGetSim()->dGetD50Coarse());
790
791 break;
792 }
793 }

797int CGeomCell::nGetLayerAtElev(double const dElev) const
798{
800 if (dElev < m_VdAllHorizonTopElev[0])
801 return ELEV_IN_BASEMENT;
802
803 for (unsigned int nLayer = 1; nLayer < m_VdAllHorizonTopElev.size();
804 nLayer++)
805 {
