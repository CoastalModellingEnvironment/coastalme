38bool bPolygonAndAdjCompare(const vector<int>& nVLeft, const vector<int>& nVRight)
39{
40 // Each row (vector) of this vector-of-vectors is:
41 // 0: This-polygon ID (in down-coast seq when sorted)
42 // 1: This-polygon down-coast (true) or up-coast (false) sediment movement
43 // 2 and subsequent: if sediment movement is down-coast, IDs of down-coast adjacent polygons; if sediment movement is up-coast, IDs of up-coast adjacent polygons
44
45 bool const bDownCoastLeft = nVLeft[1];
46 bool const bDownCoastRight = nVRight[1];
47
48 if (bDownCoastLeft)
49 {
50 // LHS polygon is down-coast. First, deal with polygon 0
51 if (nVLeft[0] == 0)
52 // Do not swap LHS and RHS polygons
53 return true;
54
55 if (nVRight[0] == 0)
56 // Swap LHS and RHS polygons
57 return false;
58
59 if ((nVLeft.size() >= 3) && (nVRight.size() >= 3))
60 {
61 // Search the adjacent polygons of the LHS argument, is there an "off edge" amongst them?
62 bool bLHSOffEdge = false;
63
64 for (unsigned int n = 2; n < nVLeft.size(); n++)
65 {
66 if (nVLeft[n] == INT_NODATA)
67 {
68 bLHSOffEdge = true;
69 break;
70 }
71 }
72
73 if (bLHSOffEdge)
74 // Yes, there is an "off edge" among the adjacent polygons of the LHS polygon: so swap LHS and RHS polygons
75 return false;
76
77 // Search the adjacent polygons of the RHS argument, is there an "off edge" amongst them?
78 bool bRHSOffEdge = false;
79
80 for (unsigned int n = 2; n < nVRight.size(); n++)
81 {
82 if (nVRight[n] == INT_NODATA)
83 {
84 bRHSOffEdge = true;
85 break;
86 }
87 }
88
89 if (bRHSOffEdge)
90 // Yes, there is an "off edge" among the adjacent polygons of the LHS polygon: do not swap LHS and RHS polygons
91 return true;
92
93 // Do we have a local change of sediment direction?
94 if (! bDownCoastRight)
95 // Yes, there is a change of sediment direction between this and the adjacent polygon. Keep the existing sequence
96 return true;
97
98 // Now sort out polygon-to-polygon dependencies. We need to put 'target' polygons after 'source' polygons, so that the source is processed before the target. So is the RHS polygon among the adjacent polygons of the LHS polygon?
99 bool bLeftFound = false;
100
101 for (unsigned int n = 2; n < nVLeft.size(); n++)
102 {
103 if (nVLeft[n] == nVRight[0])
104 bLeftFound = true;
105 }
106
107 if (bLeftFound)
108 // Yes, the RHS polygon is among the adjacent polygons of the LHS polygon, so uncons sediment movement is from the LHS polygon to the RHS polygon, and so down-coast (i.e. along the coast in the direction of increasing coastline point numbers). Keep the existing sequence
109 return true;
110
111 // Is the LHS polygon among the adjacent polygons of the RHS polygon?
112 bool bRightFound = false;
113
114 for (unsigned int n = 2; n < nVRight.size(); n++)
115 {
116 if (nVRight[n] == nVLeft[0])
117 bRightFound = true;
118 }
119
120 if (bRightFound)
121 // Yes, the LHS polygon is among the adjacent polygons of the RHS polygon, so uncons sediment movement is from the RHS polygon to the LHS polygon, and so up-coast (i.e. along the coast in the direction of decreasing coastline point numbers). Swap them
122 return false;
123 }
124
125 // Neither polygon has an "off edge", and each polygon does not have the other polygon's coast ID amongst its list of adjacent polygons. So just put the polygon in increasing own-coast sequence
126 if (nVLeft[0] < nVRight[0])
127 return true;
128 else
129 return false;
130 }
131 else
132 {
133 // LHS polygon is up-coast. First, deal with polygon 0
134 if (nVLeft[0] == 0)
135 // Swap LHS and RHS polygons
136 return false;
137
138 if (nVRight[0] == 0)
139 // Do not swap LHS and RHS polygons
140 return true;
141
142 if ((nVLeft.size() >= 3) && (nVRight.size() >= 3))
143 {
144 // Next, search the adjacent polygons of the LHS argument, is there an "off edge" amongst them?
145 bool bLHSOffEdge = false;
146
147 for (unsigned int n = 2; n < nVLeft.size(); n++)
148 {
149 if (nVLeft[n] == INT_NODATA)
150 {
151 bLHSOffEdge = true;
152 break;
153 }
154 }
155
156 if (bLHSOffEdge)
157 // Yes, there is an "off edge" among the adjacent polygons of the LHS polygon: so do not swap LHS and RHS polygons
158 return true;
159
160 // Search the adjacent polygons of the RHS argument, is there an "off edge" amongst them?
161 bool bRHSOffEdge = false;
162
163 for (unsigned int n = 2; n < nVRight.size(); n++)
164 {
165 if (nVRight[n] == INT_NODATA)
166 {
167 bRHSOffEdge = true;
168 break;
169 }
170 }
171
172 if (bRHSOffEdge)
173 // Yes, there is an "off edge" among the adjacent polygons of the LHS polygon: swap LHS and RHS polygons
174 return false;
175
176 // Do we have a local change of sediment direction?
177 if (! bDownCoastRight)
178 // Yes, there is a change of sediment direction between this and the adjacent polygon. Keep the existing sequence
179 return true;
180
181 // Now sort out polygon-to-polygon dependencies. We need to put 'target' polygons after 'source' polygons, so that the source is processed before the target. So is the RHS polygon among the adjacent polygons of the LHS polygon?
182 bool bLeftFound = false;
183
184 for (unsigned int n = 2; n < nVLeft.size(); n++)
185 {
186 if (nVLeft[n] == nVRight[0])
187 bLeftFound = true;
188 }
189
190 if (bLeftFound)
191 // Yes, the RHS polygon is among the adjacent polygons of the LHS polygon, so uncons sediment movement is from the LHS polygon to the RHS polygon, and so down-coast (i.e. along the coast in the direction of increasing coastline point numbers). Keep the existing sequence of polygons
192 return true;
193
194 // Is the LHS polygon among the adjacent polygons of the RHS polygon?
195 bool bRightFound = false;
196
197 for (unsigned int n = 2; n < nVRight.size(); n++)
198 {
199 if (nVRight[n] == nVLeft[0])
200 bRightFound = true;
201 }
202
203 if (bRightFound)
204 // Yes, the LHS polygon is among the adjacent polygons of the RHS polygon, so uncons sediment movement is from the RHS polygon to the LHS polygon, and so up-coast (i.e. along the coast in the direction of decreasing coastline point numbers). Swap the LHS and RHS polygons
205 return false;
206 }
207
208 // Neither polygon has an "off edge", and each polygon does not have the other polygon's coast ID amongst its list of adjacent polygons. So just put the polygons in decreasing own-coast sequence
209 if (nVLeft[0] < nVRight[0])
210 return false;
211 else
212 return true;
213 }
214
215 // Default return value, should never get here
216 return true;
217}

231 {
232 // Update the values of pre-existing unconsolidated sediment, for all three size classes, to include unconsolidated sediment derived from platform erosion, cliff collapse, and sediment input events

235
236 if (m_nLogFileDetail >= LOG_FILE_MIDDLE_DETAIL)
237 {
238 WritePolygonSedimentBeforeMovement();

241
242 // Here we set up a 3D vector (i.e. a vector-of-vectors-of-vectors) to sort the polygons. The innermost (1D) vector (nVPolyAndAdj) comprises these elements, for a single polygon:
243 // 0 This-polygon coast ID (in down-coast seq when sorted)
244 // 1 This-polygon down-coast (true) or up-coast (false) sediment movement
245 // 2 and subsequent: if sediment movement is down-coast, coast IDs of down-coast adjacent polygons; if sediment movement is up-coast, coast IDs of up-coast adjacent polygons
246 vector<int> nVPolyAndAdj;
247
248 // The middle-most (2D) vector comprises nVPolyAndAdj vectors, for a single coast
249 vector<vector<int>> nVVCoastPolyAndAdjacent;
250
251 // The outermost (3D) vector comprises nVVCoastPolyAndAdjacent for all coasts
252 vector<vector<vector<int>>> nVVVAllCoastPolyAndAdjacent;
253
254 // Now preparte to route actually-eroded sand/coarse sediment to adjacent polygons, or off-grid. Sort polygons first
255 for (int nCoast = 0; nCoast < nCoastSize; nCoast++)
256 {
257 // For each coast, update the values of pre-existing unconsolidated sediment, for all three size classes, to include unconsolidated sediment derived from platform erosion, cliff collapse, and sediment input events
258 AllPolygonsUpdateStoredUncons(nCoast);

260 nVVCoastPolyAndAdjacent.clear();
261 for (int nn = 0; nn < m_VCoast[nCoast].nGetNumPolygons(); nn++)
262 {
263 CGeomCoastPolygon const* pPolygon = m_VCoast[nCoast].pGetPolygon(nn);
264 nVPolyAndAdj.clear();
265
266 // The first [0] nVPolyAndAdj array item is the polygon's down-coast ID
267 nVPolyAndAdj.push_back(pPolygon->nGetPolygonCoastID());
268
269 if (pPolygon->bDownCoastThisIter())
270 {
271 // Sediment is leaving this polygon in a down-coast direction. Set this as the second [1] nVPolyAndAdj array item
272 nVPolyAndAdj.push_back(true);
273
274 // Third [2] and subsequent nVPolyAndAdj array items are the down-coast seqs of adjacent down-coast polygons
275 for (int nAdj = 0; nAdj < pPolygon->nGetNumDownCoastAdjacentPolygons(); nAdj++)
276 {
277 // Save the coast ID of each down-coast adjacent polygon
278 int const nAdjPolyID = pPolygon->nGetDownCoastAdjacentPolygon(nAdj);
279 nVPolyAndAdj.push_back(nAdjPolyID);
280 }
281 }
282 else
283 {
284 // Sediment is leaving this polygon in an up-coast direction. Set this as the second [1] nVPolyAndAdj array item
285 nVPolyAndAdj.push_back(false);
286
287 // Third [2] and susequent nVPolyAndAdj array items are the down-coast IDs of adjacent up-coast polygons
288 for (int nAdj = 0; nAdj < pPolygon->nGetNumUpCoastAdjacentPolygons(); nAdj++)
289 {
290 // Save the coast ID of each up-coast adjacent polygon
291 int const nAdjPolyID = pPolygon->nGetUpCoastAdjacentPolygon(nAdj);
292 nVPolyAndAdj.push_back(nAdjPolyID);
293 }
294 }
295 // Save info for this polygon
296 nVVCoastPolyAndAdjacent.push_back(nVPolyAndAdj);
297 }
298 // Save info for this coast
299 nVVVAllCoastPolyAndAdjacent.push_back(nVVCoastPolyAndAdjacent);

301
302 if (m_nLogFileDetail >= LOG_FILE_MIDDLE_DETAIL)
303 {
304 // Write out the unsorted polygon sequence for all coasts
305 WritePolygonUnsortedSequence(nVVVAllCoastPolyAndAdjacent);
306 }
307
308 for (int nCoast = 0; nCoast < nCoastSize; nCoast++)
309 {
310 // OK, now sort the array using bPolygonAndAdjCompare(), so that 'target' polygons are processed after 'source' polygons NOTE: crashes if just use "sort", related to this? https://stackoverflow.com/questions/18291620/why-will-stdsort-crash-if-the-comparison-function-is-not-as-operator
311 stable_sort(nVVVAllCoastPolyAndAdjacent[nCoast].begin(), nVVVAllCoastPolyAndAdjacent[nCoast].end(), bPolygonAndAdjCompare);
