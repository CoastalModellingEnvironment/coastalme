64bool CGeomILine::bIsPresent(int const nX, int const nY)
65{
66 int const nSize = static_cast<int>(m_VPoints.size());
67
68 if (nSize == 0)
69 return false;
70
71 for (int n = 0; n < nSize; n++)
72 {
73 if ((nX == m_VPoints[n].nGetX()) && (nY == m_VPoints[n].nGetY()))
74 return true;
75 }
76
77 return false;
78}
