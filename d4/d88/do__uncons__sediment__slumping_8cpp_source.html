51struct UnstableCell
52{
53 int nX;
54 int nY;
55 double dInstability;
56
58 UnstableCell(int const x, int const y, double const instability)
59 : nX(x), nY(y), dInstability(instability)
60 {
61 }
62};

68struct UnstableCellComparator
69{
70 bool operator()(UnstableCell const& a, UnstableCell const& b) const
71 {
72 // Return true if a has LOWER priority than b. This makes the priority queue a max-heap (highest instability first)
73 return a.dInstability < b.dInstability;
74 }
75};

81double CSimulation::dCalcSlopeForUnconsSlumping(int const nX1, int const nY1, int const nX2, int const nY2) const
82{
83 // Get top surface elevations
84 double const dElev1 = m_pRasterGrid->m_Cell[nX1][nY1].dGetAllSedTopElevOmitTalus();
85 double const dElev2 = m_pRasterGrid->m_Cell[nX2][nY2].dGetAllSedTopElevOmitTalus();
86
87 // Calculate distance between cells
88 double dDistance;
89 if ((nX1 != nX2) && (nY1 != nY2))
90 {
91 // Diagonal neighbor: distance is sqrt(2) * cell size
92 dDistance = SQRT2 * m_dCellSide;
93 }
94 else
95 {
96 // Orthogonal neighbor: distance is cell size
97 dDistance = m_dCellSide;
98 }
99
100 // Calculate slope as rise over run (tangent of angle)
101 double const dRise = dElev1 - dElev2;
102 double const dSlope = dRise / dDistance;
103
104 return dSlope;
105}

110double CSimulation::dCalculateSlumpInstability(int const nX, int const nY) const
111{
112 double dMaxInstability = 0.0;
113
114 // Check slopes to all 8 neighbors
115 for (int di = -1; di <= 1; di++)
116 {
117 for (int dj = -1; dj <= 1; dj++)
118 {
119 if (di == 0 && dj == 0)
120 continue; // Skip self
121
122 int const nNeighbourX = nX + di;
123 int const nNeighbourY = nY + dj;
124
125 // Check if neighbour is within grid bounds
126 if (nNeighbourX < 0 || nNeighbourX >= m_nXGridSize || nNeighbourY < 0 || nNeighbourY >= m_nYGridSize)
127 continue;
128
129 // Calculate slope to this neighbor
130 double const dSlope = dCalcSlopeForUnconsSlumping(nX, nY, nNeighbourX, nNeighbourY);
131
132 // Calculate instability (excess slope beyond angle of repose)
133 double const dInstability = dSlope - TAN_ANGLE_OF_REPOSE;
134
135 // Track maximum instability
136 if (dInstability > dMaxInstability)
137 dMaxInstability = dInstability;
138 }
139 }
140
141 return dMaxInstability;
142}

147set<pair<int, int>> CSimulation::prDoSlumpRedistributeSediment(int const nX, int const nY)
148{
149 set<pair<int, int>> prAffectedNeighbours;
150
151 // Check if source cell has sediment layers
152 int const nTopLayer = m_pRasterGrid->m_Cell[nX][nY].nGetTopNonZeroLayerAboveBasement();
153 if (nTopLayer <= 0)
154 return prAffectedNeighbours; // No layers to redistribute
155
156 // Get source cell elevation
157 double const dSourceElev = m_pRasterGrid->m_Cell[nX][nY].dGetAllSedTopElevOmitTalus();
158
159 // Get unconsolidated sediment depths from top layer
160 CRWCellSediment* pUnconsolidated = m_pRasterGrid->m_Cell[nX][nY].pGetLayerAboveBasement(nTopLayer)->pGetUnconsolidatedSediment();
161 double dSand = pUnconsolidated->dGetSandDepth();
162 double dCoarse = pUnconsolidated->dGetCoarseDepth();
163 double dFine = pUnconsolidated->dGetFineDepth();
164
165 double const dTotalDepth = dSand + dCoarse + dFine;
166
167 // Check if there's enough sediment to redistribute
168 if (dTotalDepth < MIN_SLUMP_VOLUME)
169 return prAffectedNeighbours;
170
171 // Find downslope neighbours and calculate weights
172 struct Neighbour
173 {
174 int x;
175 int y;
176 double slope;
177 double weight;
178 };
179
180 vector<Neighbour> VNeighbourDownslope;
181 double dTotalWeight = 0.0;
182
183 for (int di = -1; di <= 1; di++)
184 {
185 for (int dj = -1; dj <= 1; dj++)
186 {
187 if (di == 0 && dj == 0)
188 continue;
189
190 int const nNeighbourX = nX + di;
191 int const nNeighbourY = nY + dj;
192
193 // Check bounds
194 if (nNeighbourX < 0 || nNeighbourX >= m_nXGridSize || nNeighbourY < 0 || nNeighbourY >= m_nYGridSize)
195 continue;
196
197 // Get neighbour's elevation
198 double const dNeighborElev = m_pRasterGrid->m_Cell[nNeighbourX][nNeighbourY].dGetAllSedTopElevOmitTalus();
199
200 // Only redistribute to lower neighbours if slope is above the stability threshold
201 if (dNeighborElev < dSourceElev)
202 {
203 double const dSlope = dCalcSlopeForUnconsSlumping(nX, nY, nNeighbourX, nNeighbourY);
204
205 if (dSlope > TAN_ANGLE_OF_REPOSE)
206 {
207 // Weight proportional to excess slope
208 double const dWeight = dSlope - TAN_ANGLE_OF_REPOSE;
209 VNeighbourDownslope.push_back({nNeighbourX, nNeighbourY, dSlope, dWeight});
210 dTotalWeight += dWeight;
211 }
212 }
213 }
214 }
215
216 // No valid downslope neighbors
217 if (VNeighbourDownslope.empty())
218 return prAffectedNeighbours;
219
220 // Calculate amount to redistribute
221 double const dAmountToMove = dTotalDepth * SLUMP_REDISTRIBUTION_FRACTION;
222
223 // Track remaining sediment in source cell
224 double dRemainingSand = dSand;
225 double dRemainingCoarse = dCoarse;
226 double dRemainingFine = dFine;
227
228 // Distribute proportionally to neighbours
229 for (auto const& neighbour : VNeighbourDownslope)
230 {
231 double const dFraction = neighbour.weight / dTotalWeight;
232 double const dMoveDepth = dAmountToMove * dFraction;
233
234 // Calculate amount of each size class to move (proportional to composition)
235 double const dMoveSand = dMoveDepth * (dSand / dTotalDepth);
236 double const dMoveCoarse = dMoveDepth * (dCoarse / dTotalDepth);
237 double const dMoveFine = dMoveDepth * (dFine / dTotalDepth);
238
239 // Update remaining amounts
240 dRemainingSand -= dMoveSand;
241 dRemainingCoarse -= dMoveCoarse;
242 dRemainingFine -= dMoveFine;
243
244 // Track avalanche deposition (total depth moved into this cell)
245 int const nNeighborTopLayer = m_pRasterGrid->m_Cell[neighbour.x][neighbour.y].nGetTopNonZeroLayerAboveBasement();
246
247 if (nNeighborTopLayer >= 0)
248 {
249 CRWCellSediment* pNeighborUnconsolidated = m_pRasterGrid->m_Cell[neighbour.x][neighbour.y].pGetLayerAboveBasement(nNeighborTopLayer)->pGetUnconsolidatedSediment();
250
251 pNeighborUnconsolidated->AddSandDepth(dMoveSand);
252 pNeighborUnconsolidated->AddCoarseDepth(dMoveCoarse);
253 pNeighborUnconsolidated->AddFineDepth(dMoveFine);
254
255 // Recalculate neighbour elevations
256 m_pRasterGrid->m_Cell[neighbour.x][neighbour.y].CalcAllLayerElevsAndD50();
257 m_pRasterGrid->m_Cell[neighbour.x][neighbour.y].SetSeaDepth();
258
259 // Track slumping deposition (total depth moved into this cell)
260 m_pRasterGrid->m_Cell[neighbour.x][neighbour.y].IncrSlumpDeposition(dMoveDepth);
261
262 // Debug output
263 if (dMoveDepth > 0.001)
264 {
265 LogStream << " Avalanche: moved " << dMoveDepth << " m from (" << nX << "," << nY << ") to (" << neighbour.x << "," << neighbour.y << "), cell now has " << m_pRasterGrid->m_Cell[neighbour.x][neighbour.y].dGetAvalancheDeposition() << " m" << endl;
266 }
267
268 // Track affected neighbor
269 prAffectedNeighbours.insert(std::make_pair(neighbour.x, neighbour.y));
270 SlumpMarkCellDirty(neighbour.x, neighbour.y);
271 }
272 }
273
274 // Update source cell with remaining sediment
275 pUnconsolidated->SetSandDepth(dRemainingSand);
276 pUnconsolidated->SetCoarseDepth(dRemainingCoarse);
277 pUnconsolidated->SetFineDepth(dRemainingFine);
278
279 // Recalculate source cell elevations
280 m_pRasterGrid->m_Cell[nX][nY].CalcAllLayerElevsAndD50();
281 m_pRasterGrid->m_Cell[nX][nY].SetSeaDepth();
282
283 return prAffectedNeighbours;
284}

289int CSimulation::nDoSedimentSlumping(void)
290{
291 // If no cells changed, nothing to do
292 if (m_prSlumpDirtyCells.empty())
293 return RTN_OK;
294
295 if (m_nLogFileDetail >= LOG_FILE_MIDDLE_DETAIL)
296 {
297 LogStream << m_ulIter << ": Processing sediment avalanches on " << m_prSlumpDirtyCells.size() << " changed cells" << endl;
298 }
299
300 // Build initial candidate set: dirty cells plus their eight neighbours
301 set<pair<int, int>> prCandidateCells;
302
303 for (auto const& dirtyCell : m_prSlumpDirtyCells)
304 {
305 prCandidateCells.insert(dirtyCell);
306
307 // Add the eight neighbours
308 for (int di = -1; di <= 1; di++)
309 {
310 for (int dj = -1; dj <= 1; dj++)
311 {
312 int const nNeighbourX = dirtyCell.first + di;
313 int const nNeighbourY = dirtyCell.second + dj;
314
315 if (nNeighbourX >= 0 && nNeighbourX < m_nXGridSize &&
316 nNeighbourY >= 0 && nNeighbourY < m_nYGridSize)
317 {
318 prCandidateCells.insert(make_pair(nNeighbourX, nNeighbourY));
319 }
320 }
321 }
322 }
323
324 // Build priority queue of unstable cells
325 priority_queue<UnstableCell, vector<UnstableCell>, UnstableCellComparator> pq;
326
327 for (auto const& cell : prCandidateCells)
328 {
329 double const dInstability = dCalculateSlumpInstability(cell.first, cell.second);
330 if (dInstability > 0.0)
331 {
332 pq.push(UnstableCell(cell.first, cell.second, dInstability));
333 }
334 }
335
336 // Process queue until empty or max iterations reached
337 int nIterations = 0;
338 int nCellsProcessed = 0;
339 set<pair<int, int>> prProcessedCells;
340
341 while (!pq.empty() && nIterations < MAX_SLUMP_ITERATIONS)
342 {
343 UnstableCell const unstable = pq.top();
344 pq.pop();
345
346 // Skip if already processed (may be in queue multiple times)
347 pair<int, int> const cell_coords = make_pair(unstable.nX, unstable.nY);
348 if (prProcessedCells.count(cell_coords) > 0)
349 continue;
350
351 // Redistribute sediment
352 set<pair<int, int>> const prAffectedNeighbours = prDoSlumpRedistributeSediment(unstable.nX, unstable.nY);
353
354 // Mark as processed
355 prProcessedCells.insert(cell_coords);
356 nCellsProcessed++;
357
358 // Check affected neighbours for new instability
359 for (auto const& neighbour : prAffectedNeighbours)
360 {
361 if (prProcessedCells.count(neighbour) == 0)
362 {
363 double const dNewInstability = dCalculateSlumpInstability(neighbour.first, neighbour.second);
364 if (dNewInstability > 0.0)
365 {
366 pq.push(UnstableCell(neighbour.first, neighbour.second, dNewInstability));
367 }
368 }
369 }
370
371 nIterations++;
372 }
373
374 // Log results
375 if (m_nLogFileDetail >= LOG_FILE_MIDDLE_DETAIL)
376 {
377 LogStream << " Processed " << nCellsProcessed << " unstable cells in " << nIterations << " iterations" << endl;
378 }
379
380 // Warn if we hit max iterations
381 if (nIterations >= MAX_SLUMP_ITERATIONS)
382 {
383 LogStream << WARN << m_ulIter << ": Sediment avalanching hit maximum iteration limit (" << MAX_SLUMP_ITERATIONS << ")" << endl;
384 }
385
386 // Note: m_prSlumpDirtyCells is NOT cleared here, so it remains available for GIS output. It will be cleared at the start of the next timestep
387
388 return RTN_OK;
389}
